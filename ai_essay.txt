# Initialize the sentiment analyzer once
analyzer = SentimentIntensityAnalyzer()

# Enhanced analysis function
def analyze_essay(essay, label):
    # ... (rest of your function definition is correct) ...
    print(f"--- {label} Essay Analysis ---")

    # Readability
    print(f"Flesch Reading Ease: {textstat.flesch_reading_ease(essay):.2f}") # Added formatting
    print(f"Flesch-Kincaid Grade Level: {textstat.flesch_kincaid_grade(essay):.2f}")

    # Length / Basic Stats
    words = essay.split() # Simple split for word list
    word_count = len(words) # Use basic len() after splitting
    sentence_count = textstat.sentence_count(essay)
    print(f"Word Count: {word_count}")
    print(f"Sentence Count: {sentence_count}")
    if sentence_count > 0:
      print(f"Average Sentence Length: {word_count / sentence_count:.2f} words")
    else:
      print("Average Sentence Length: N/A (No sentences found)")


    # Lexical Diversity
    # Using NLTK's word_tokenize for potentially better tokenization than simple split()
    try:
        tokens = nltk.word_tokenize(essay.lower()) # Tokenize and convert to lowercase
        unique_tokens = set(tokens)
        if len(tokens) > 0:
            lexical_diversity = len(unique_tokens) / len(tokens)
            print(f"Lexical Diversity (Type-Token Ratio): {lexical_diversity:.3f}")
        else:
            print("Lexical Diversity: N/A (No tokens found)")
    except LookupError: # Catch potential error if punkt wasn't successfully downloaded/found
